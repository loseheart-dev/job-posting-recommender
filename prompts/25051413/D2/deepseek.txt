下面给出一个 Python 代码框架，展示如何实现"独立测试数据 + 三类分析输出结构"作为算法模块的基础核心。代码不要求完整，但会突出关键部分和注释。

```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import classification_report, roc_auc_score, confusion_matrix
from sklearn.inspection import permutation_importance


# ============================================================
# 1. 独立测试数据的创建（核心要求：测试集全程隔离）
# ============================================================
def load_and_split_data(data_path, target_col, test_size=0.2, random_state=42):
    """
    加载数据并严格划分训练集和独立测试集。
    独立测试集在后续任何预处理、特征工程、调参中都不允许被使用。
    """
    df = pd.read_csv(data_path)
    X = df.drop(columns=[target_col])
    y = df[target_col]

    # 划分训练集和独立测试集
    X_train, X_test, y_train, y_test = train_test_split(
        X, y, test_size=test_size, random_state=random_state, stratify=y
    )

    # 这里可以保存测试集到单独文件，防止意外修改
    # X_test.to_csv('independent_test_X.csv', index=False)
    # y_test.to_csv('independent_test_y.csv', index=False)

    return X_train, X_test, y_train, y_test


# ============================================================
# 2. 三类分析输出结构
# ============================================================
class AlgorithmAnalysisPipeline:
    """
    算法模块的基础分析框架，包含三类输出：
    1. 描述性分析输出（Descriptive）
    2. 诊断性分析输出（Diagnostic）
    3. 预测/建议性分析输出（Predictive/Prescriptive）
    """

    def __init__(self, X_train, y_train, X_test, y_test):
        self.X_train = X_train
        self.y_train = y_train
        self.X_test = X_test
        self.y_test = y_test
        self.model = None
        self.feature_importances = None
        self.test_predictions = None

    # ---------- 1. 描述性分析 ----------
    def descriptive_analysis(self):
        """
        输出数据的基本描述统计、分布情况、缺失值、异常值等。
        目的：了解数据现状，发现潜在质量问题。
        """
        print("=" * 60)
        print("1. 描述性分析输出")
        print("=" * 60)

        # 训练集基本信息
        print(f"训练集样本数: {self.X_train.shape[0]}, 特征数: {self.X_train.shape[1]}")
        print(f"测试集样本数: {self.X_test.shape[0]}, 特征数: {self.X_test.shape[1]}")

        # 目标变量分布
        print("\n目标变量分布（训练集）:")
        print(self.y_train.value_counts(normalize=True))

        # 数值特征统计描述
        print("\n数值特征统计描述（训练集）:")
        print(self.X_train.describe().T[['mean', 'std', 'min', 'max']])

        # 缺失值统计
        missing = self.X_train.isnull().sum()
        missing_pct = (missing / len(self.X_train)) * 100
        missing_df = pd.DataFrame({'缺失数量': missing, '缺失比例(%)': missing_pct})
        missing_df = missing_df[missing_df['缺失数量'] > 0].sort_values('缺失比例(%)', ascending=False)
        if not missing_df.empty:
            print("\n缺失值情况（训练集）:")
            print(missing_df)
        else:
            print("\n无缺失值。")

        # 可视化示例：目标变量分布和特征分布（这里只展示，实际可保存图片）
        # plt.figure(figsize=(10,4))
        # plt.subplot(1,2,1)
        # self.y_train.value_counts().plot(kind='bar')
        # plt.title('目标变量分布')
        # plt.subplot(1,2,2)
        # self.X_train.hist(figsize=(10,8))
        # plt.tight_layout()
        # plt.show()

    # ---------- 2. 诊断性分析 ----------
    def diagnostic_analysis(self, model=None):
        """
        在训练集上训练模型，并输出特征重要性、系数、相关性等诊断信息。
        目的：理解哪些因素驱动预测，解释模型行为。
        """
        print("\n" + "=" * 60)
        print("2. 诊断性分析输出")
        print("=" * 60)

        if model is None:
            # 默认使用随机森林，也可替换为其他模型
            model = RandomForestClassifier(n_estimators=100, random_state=42, n_jobs=-1)

        # 训练模型（注意：这里只用训练集）
        model.fit(self.X_train, self.y_train)
        self.model = model

        # 特征重要性（基于随机森林或排列重要性）
        if hasattr(model, 'feature_importances_'):
            importances = model.feature_importances_
        else:
            # 对于不支持 feature_importances_ 的模型，使用排列重要性
            perm_importance = permutation_importance(model, self.X_train, self.y_train, n_repeats=10, random_state=42)
            importances = perm_importance.importances_mean

        feature_imp = pd.DataFrame({
            'feature': self.X_train.columns,
            'importance': importances
        }).sort_values('importance', ascending=False)

        print("\n特征重要性 Top 10:")
        print(feature_imp.head(10))

        # 相关性分析：特征与目标变量的相关系数（数值特征）
        numeric_cols = self.X_train.select_dtypes(include=[np.number]).columns
        if len(numeric_cols) > 0:
            corr_with_target = self.X_train[numeric_cols].apply(lambda x: x.corr(self.y_train))
            corr_df = pd.DataFrame({'feature': numeric_cols, 'correlation': corr_with_target}).sort_values('correlation', key=abs, ascending=False)
            print("\n与目标变量相关性最高的特征 Top 10:")
            print(corr_df.head(10))

        # 可视化特征重要性（可选）
        # plt.figure(figsize=(10,6))
        # sns.barplot(x='importance', y='feature', data=feature_imp.head(15))
        # plt.title('Feature Importance')
        # plt.tight_layout()
        # plt.show()

    # ---------- 3. 预测/建议性分析 ----------
    def predictive_analysis(self, threshold=0.5):
        """
        在独立测试集上进行预测，输出评估指标、混淆矩阵等。
        并根据业务场景提供建议（如阈值调整、风险分层）。
        目的：评估模型真实效果，指导业务决策。
        """
        print("\n" + "=" * 60)
        print("3. 预测/建议性分析输出（独立测试集）")
        print("=" * 60)

        if self.model is None:
            raise ValueError("请先运行 diagnostic_analysis 训练模型。")

        # 在独立测试集上预测
        y_pred_proba = self.model.predict_proba(self.X_test)[:, 1]
        y_pred = (y_pred_proba >= threshold).astype(int)
        self.test_predictions = y_pred

        # 评估指标
        print(f"\n阈值 = {threshold} 时的评估指标：")
        print(classification_report(self.y_test, y_pred))
        print(f"AUC-ROC: {roc_auc_score(self.y_test, y_pred_proba):.4f}")

        # 混淆矩阵
        cm = confusion_matrix(self.y_test, y_pred)
        print("\n混淆矩阵:")
        print(cm)

        # 业务建议示例：根据预测概率进行分层
        risk_levels = pd.cut(y_pred_proba, bins=[0, 0.3, 0.6, 1.0], labels=['低风险', '中风险', '高风险'])
        risk_summary = pd.DataFrame({
            '真实标签': self.y_test.values,
            '预测概率': y_pred_proba,
            '风险等级': risk_levels
        })
        print("\n风险分层统计（测试集）:")
        print(risk_summary.groupby('风险等级').agg(
            样本数=('预测概率', 'count'),
            平均真实标签=('真实标签', 'mean'),
            平均预测概率=('预测概率', 'mean')
        ))

        # 建议：针对高风险群体优先干预等
        print("\n业务建议：")
        print("- 高风险群体可优先进行人工审核或干预。")
        print("- 可根据业务目标调整阈值以平衡精确率与召回率。")

        # 可视化 ROC 曲线（可选）
        # from sklearn.metrics import roc_curve
        # fpr, tpr, _ = roc_curve(self.y_test, y_pred_proba)
        # plt.plot(fpr, tpr)
        # plt.xlabel('False Positive Rate')
        # plt.ylabel('True Positive Rate')
        # plt.title('ROC Curve (Test Set)')
        # plt.show()


# ============================================================
# 主流程示例
# ============================================================
if __name__ == "__main__":
    # 假设数据路径和标签列
    DATA_PATH = "your_data.csv"
    TARGET_COL = "target"

    # 1. 加载数据并划分独立测试集
    X_train, X_test, y_train, y_test = load_and_split_data(DATA_PATH, TARGET_COL)

    # 2. 初始化分析管线
    pipeline = AlgorithmAnalysisPipeline(X_train, y_train, X_test, y_test)

    # 3. 执行三类分析输出
    pipeline.descriptive_analysis()      # 描述性输出
    pipeline.diagnostic_analysis()       # 诊断性输出（同时训练模型）
    pipeline.predictive_analysis()       # 预测/建议性输出（使用独立测试集）
```

关键点说明

独立测试数据
通过 train_test_split 划分出 X_test 和 y_test，后续所有预处理和训练都只使用训练集。
测试集在整个过程中只用于最终预测和评估（predictive_analysis），绝不允许用于特征选择、调参或模型训练。
为了更严格，可以将测试集单独保存，防止代码中意外引用。

三类分析输出结构
描述性分析：提供数据全貌，包括样本量、目标分布、统计特征、缺失值等，帮助团队理解数据现状。
诊断性分析：在训练集上训练模型，输出特征重要性、相关系数等，解释模型决策依据，定位关键因素。
预测/建议性分析：在独立测试集上评估模型，输出分类报告、AUC、混淆矩阵，并给出风险分层和业务建议，使结果可落地。

这样的结构保证了算法模块的可信度、可解释性和业务可用性，符合团队协作要求。
